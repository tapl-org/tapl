\section{Syntax}\label{sec:syntax}

\[
\begin{array}{rcll}
t, F & ::= & & \textit{terms} \\
  & \mid & x & \textit{variable} \\
  & \mid & \lam{x}{t} & \textit{abstraction} \\
  & \mid & \lamF{x}{F}{t} & \textit{abstraction with a form} \\
  & \mid & t\ t & \textit{application} \\
  & \mid & \{l_1 = t_1, \dots, l_n = t_n\} & \textit{struct} \\
  & \mid & t.l & \textit{projection} \\
  & \mid & [b] & \textit{byte array} \\
  & \mid & \fm{t}{F} & \textit{formed term} \\
  & \mid & F \to F & \textit{function form}
\end{array}
\]
\begin{itemize}
\item $t$ ranges over terms, $x$ over variables, $l$ over labels.
\item Forms are terms too, written $F, G$.
\item $b$ ranges over byte sequences, written in hex: $\bytes{2a000000}$;
  $[\,]$ is the empty byte array.
\item Struct fields are ordered; a label may be empty, and labels may repeat.
\end{itemize}

\section{Computation (at the root)}\label{sec:computation}

\begin{rules}
\axiomrow{E-Beta}
  {(\lam{x}{t_1})\ t_2 \;\step\; \subst{x}{t_2}{t_1}}
\axiomrow{E-BetaForm}
  {(\lamF{x}{F}{t_1})\ t_2 \;\step\; \subst{x}{\fm{\fn{strip}(t_2)}{F}}{t_1}}
\rulerow{E-Proj}
  {k = \min\{\, i \mid l_i = l \,\}}
  {\{l_i = t_i\}.l \;\step\; t_k}
\rulerow{E-ProjForm}
  {k = \min\{\, i \mid l_i = l \,\} \\ m = \min\{\, j \mid l'_j = l \,\}}
  {(\fm{\{l_i = t_i\}}{\{l'_j = F_j\}}).l \;\step\; \fm{\fn{strip}(t_k)}{F_m}}
\rulerow{E-Select}
  {n \ge 1 \\ c \in \{[b], \fm{[b]}{F}\} \\ b \text{ is not empty} \\
   i = \min(\fn{le}(b),\, n-1)}
  {\{l_0 = t_0, \dots, l_{n-1} = t_{n-1}\}\ c \;\step\; t_i}
\end{rules}
\begin{itemize}
\item $t \step t'$ means $t$ takes one step to $t'$: by a computation rule at
  the root (this section), or else by a congruence rule that steps the part
  blocking the root (Section~\ref{sec:congruence}).
\item $\subst{x}{t_2}{t_1}$ is $t_1$ with $t_2$ put in for $x$.
\item $\fn{strip}$ drops one form: $\fn{strip}(\fm{t}{G}) = t$, and
  $\fn{strip}(t) = t$ otherwise.
\item $\fn{le}(b)$ is $b$ read as an unsigned little-endian integer: the first
  byte is the lowest. For $\bytes{0201}$,
  $\fn{le}(\mathtt{0201}) = \mathtt{02} + \mathtt{01} \cdot 256 = 258$.
\item \rn{E-Proj} and \rn{E-ProjForm} take the first field with the label.
\item \rn{E-Select} ignores labels, and a form on the index. An index past the
  last field takes the last one, so the last field is the default:
  $\{e, t\}\ \bytes{02}$ and $\{e, t\}\ \bytes{0100}$ both give $t$.
\end{itemize}

%% ── Congruence ─────────────────────────────────────────────────────────

\section{Congruence (reduce the blocker)}\label{sec:congruence}

\judgment{t \step t'}{one step}
\begin{rules}
\rulerow{E-App1}
  {t_1 \step t_1'}
  {t_1\ t_2 \;\step\; t_1'\ t_2}
\rulerow{E-SelectArg}
  {t_2 \step t_2'}
  {\{l_i = t_i\}\ t_2 \;\step\; \{l_i = t_i\}\ t_2'}
\rulerow{E-Proj1}
  {t \step t'}
  {t.l \;\step\; t'.l}
\rulerow{E-Formed}
  {t \step t'}
  {\fm{t}{F} \;\step\; \fm{\fn{strip}(t')}{F}}
\end{rules}
\begin{itemize}
\item Apply only when no computation rule applies at the root.
\item \rn{E-Formed} never reduces the form $F$; when $t$ steps to a formed term,
  $F$ replaces its form.
\item No congruence rules for: the body of a $\lambda$, a form, a struct field,
  an ordinary argument.
\end{itemize}

\section{All Combinations}\label{sec:combinations}

Every term, by its root, with the rule that applies or why none does. A term no
rule applies to is a \emph{normal form}: a variable, a value
($\lam{x}{t}$, $\lamF{x}{F}{t}$, a struct, $[b]$, $F \to G$), a formed normal
form, or a \emph{stuck} term. In the tables, ``steps'' means a rule applies to
that part.

\subsection{Terms with no rule at the root}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Term} & \textbf{Result} \\
\midrule
$x$ & normal form (variable) \\
$\lam{x}{t}$, \ $\lamF{x}{F}{t}$ & normal form (value) \\
$\{l_1 = t_1, \dots, l_n = t_n\}$ & normal form (value); fields are not reduced \\
$[b]$ & normal form (value) \\
$F \to G$ & normal form (value) \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Application \texorpdfstring{$t_1\ t_2$}{t1 t2}}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{$t_1$} & \textbf{$t_2$} & \textbf{Result} \\
\midrule
$\lam{x}{t}$ & any & \rn{E-Beta} \\
$\lamF{x}{F}{t}$ & any & \rn{E-BetaForm} \\
\midrule
$\{l_0 = t_0, \dots, l_{n-1} = t_{n-1}\}$, $n \ge 1$
  & $[b]$ or $\fm{[b]}{F}$, $b$ not empty & \rn{E-Select} \\
  & $[\,]$ or $\fm{[\,]}{F}$ & stuck \\
  & steps & \rn{E-SelectArg} \\
  & any other normal form & stuck \\
$\{\}$ & steps & \rn{E-SelectArg} \\
  & normal form & stuck \\
\midrule
$s\ s'$, \ $s.l$, \ $\fm{s}{F}$ & --- & \rn{E-App1} if $t_1$ steps, else stuck \\
\midrule
$x$ & any & stuck \\
$[b]$ & any & stuck \\
$F \to G$ & any & stuck \\
\bottomrule
\end{tabular}
\end{center}

\noindent
A formed abstraction does not apply: $(\fm{(\lam{x}{t})}{F})\ t_2$ is stuck,
since $\lam{x}{t}$ does not step.

\subsection{Projection \texorpdfstring{$t.l$}{t.l}}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{$t$} & \textbf{Case} & \textbf{Result} \\
\midrule
$\{l_i = t_i\}$ & some $l_i = l$ & \rn{E-Proj} \\
  & no $l_i = l$ & stuck \\
\midrule
$\fm{\{l_i = t_i\}}{\{l'_j = F_j\}}$ & some $l_i = l$ and some $l'_j = l$
  & \rn{E-ProjForm} \\
  & otherwise & stuck \\
\midrule
$t_1\ t_2$, \ $t'.l'$, \ other $\fm{t'}{F}$ & --- & \rn{E-Proj1} if $t$ steps, else stuck \\
\midrule
$x$ & --- & stuck \\
$\lam{x}{t'}$, \ $\lamF{x}{F}{t'}$, \ $[b]$, \ $F \to G$ & --- & stuck \\
\bottomrule
\end{tabular}
\end{center}

\noindent
In the formed row, a struct with a form that is not a struct literal (for
example $\fm{\{l = t\}}{x}$) is ``other $\fm{t'}{F}$'': it does not step, so the
projection is stuck.

\subsection{Formed term \texorpdfstring{$\fm{t}{F}$}{t\textasciitilde F}}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{$t$} & \textbf{Result} \\
\midrule
steps & \rn{E-Formed} \\
normal form & normal form; stuck if $t$ is stuck \\
\bottomrule
\end{tabular}
\end{center}

\noindent
A form on a formed normal form stays: $\fm{(\fm{[b]}{G})}{F}$ does not step.

\section{Evaluation Strategy}\label{sec:strategy}

The rules above give this strategy:
\begin{itemize}
\item \textbf{Weak:} never goes under a $\lambda$ or into a struct field.
\item \textbf{Lazy:} never reduces an ordinary argument.
\end{itemize}

%% ── Syntax sugar ───────────────────────────────────────────────────────

\section{Syntax Sugar}\label{sec:sugar}

Sugar adds no terms: each form on the left is the term on the right.
$\epsilon$ is the empty label.
\[
\begin{array}{rcll}
\kw{if}\ c\ \kw{then}\ t\ \kw{else}\ e & \equiv & \{e, t\}\ c
  & \textit{if} \\
\kw{let}\ x = t_1\ \kw{in}\ t_2 & \equiv & (\lam{x}{t_2})\ t_1
  & \textit{let} \\
\kw{let}\ \fm{x}{F} = t_1\ \kw{in}\ t_2 & \equiv & (\lamF{x}{F}{t_2})\ t_1
  & \textit{let with a form} \\
\{\dots, t, \dots\} & \equiv & \{\dots, \epsilon = t, \dots\}
  & \textit{unlabeled field}
\end{array}
\]
So $\kw{if}$ is \rn{E-Select} on two fields: $\bytes{00}$ gives the else arm,
any other non-empty index gives the then arm.

\section{Under Binders: \texorpdfstring{$\fn{shape}$}{shape}}\label{sec:shape}

$t \eval w$ means: apply $\step$ to $t$ until no rule applies; $w$ is the normal
form reached (Section~\ref{sec:combinations}).

$\fn{shape}$ is the only part of evaluation that goes under a $\lambda$ or into
a struct field, and only at the positions of the program skeleton.
\begin{itemize}
\item Each position first applies $\eval$.
\item If the result doesn't have the expected shape, $\fn{shape}$ leaves it as
  is and translation reports the error.
\end{itemize}

\paragraph{Skeleton.}
\[
\begin{array}{lll}
\fn{program}(t)  & t \eval \lam{m}{u}      & \implies \lam{m}{\fn{module}(u)} \\
\fn{module}(u)   & u \eval \{l_i = f_i\}   & \implies \{l_i = \fn{function}(f_i)\} \\
\fn{function}(f) & f \eval \lam{k}{g}      & \implies \lam{k}{\fn{blocks}(g)}
  \quad \text{(a data value: only $\eval$)} \\
\fn{blocks}(g)   & g \eval \{l_i = s_i\}   & \implies \{l_i = \fn{block}(s_i)\} \\
\fn{block}(s)    & s \eval \lam{a}{t}      & \implies \lam{a}{\fn{body}(t)}
\end{array}
\]

\paragraph{Body.}\mbox{}
\begin{rules}
\rulerow{S-Let}
  {v \eval w \\ w \text{ is an op}}
  {\fn{body}((\lam{x}{t})\ v) = (\lam{x}{\fn{body}(t)})\ \fn{op}(w)}
\rulerow{S-Inline}
  {v \eval w \\ w \text{ is not an op}}
  {\fn{body}((\lam{x}{t})\ v) = \fn{body}(\subst{x}{w}{t})}
\end{rules}
\begin{itemize}
\item \rn{S-Inline} substitutes by \rn{E-Beta}/\rn{E-BetaForm}.
\item Otherwise $\fn{body}(t) = \fn{body}(t')$ if $t \step t'$, else
  $\fn{terminal}(t)$.
\end{itemize}

\paragraph{Terminals and operands.}
\[
\begin{array}{lcll}
\fn{terminal}(f.l\ a)           & = & f.l\ \fn{arg}(a) & \textit{jump} \\
\fn{terminal}(\{l_i = t_i\}\ c) & = & \{l_i = \fn{jump}(t_i)\}\ \fn{atom}(c)
  & \textit{branch or switch} \\
\fn{terminal}(t)                & = & t & [b],\ \kw{args}.l,\ x,\ \dots \\[6pt]
\fn{op}(t\ a)                   & = & t\ \fn{arg}(a) & \\
\fn{op}(\{l_i = t_i\})          & = & \{l_i = \fn{atom}(t_i)\} & \\
\fn{op}(t)                      & = & t & \textit{a projection} \\[6pt]
\fn{arg}(a)  & = & \{l_i = \fn{atom}(t_i)\} \text{ if } a \eval \{l_i = t_i\};
  \text{ else } a\eval & \\
\fn{jump}(t) & = & f.l\ \fn{arg}(a) \text{ if } t \eval f.l\ a;
  \text{ else } t\eval & \\
\fn{atom}(t) & = & t\eval &
\end{array}
\]

\noindent
An \textbf{op} is one of:
\begin{itemize}
\item an application $t\ a$ where $t$ is not a struct ($\{\dots\}\ c$ is a
  branch or a switch);
\item a struct;
\item a projection other than $\kw{args}.l$.
\end{itemize}

\noindent
Terminal $\{t_0, \dots, t_{n-1}\}\ c$:
\begin{itemize}
\item $c$ didn't reduce to a byte array, so it's a run-time selector.
\item Labels are ignored; the field is picked as in \rn{E-Select}, so a $c$ past
  the last field takes $t_{n-1}$.
\item 2 fields ($\kw{if}\ c\ \kw{then}\ t\ \kw{else}\ e$): a branch.
\item 3+ fields: a switch whose default is the last field.
\item Translation rejects $\{\}\ c$, $\{t\}\ c$, and a field that isn't a jump.
\end{itemize}

\noindent
$\fn{shape}$ leaves every other position unreduced: a let-bound $\lambda$, a
lambda passed as an argument, a struct field outside the skeleton, and any form.
