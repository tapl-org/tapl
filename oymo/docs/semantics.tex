%% ── Goal ───────────────────────────────────────────────────────────────

\section{Goal}\label{sec:goal}

\begin{points}
\item Oymomo is an extension of the untyped $\lambda$-calculus: to variables, abstraction and
  application it adds structs, projection and byte arrays, which name
  opcodes, and a fixed point for recursion; its $\beta$ rule is call-by-value: a function is applied only to a value, so a term that still has to run is never copied or dropped.
\item Section~\ref{sec:goal} states the goal, Section~\ref{sec:syntax} gives the syntax, Section~\ref{sec:sugar} the sugar,
  Sections~\ref{sec:computation} and~\ref{sec:congruence} the reduction, Section~\ref{sec:strategy} how to reach a normal form, Section~\ref{sec:banf} ANF (A-normal form) and functions, Section~\ref{sec:modules} ingots, and Section~\ref{sec:llvm} the translation to LLVM.
\item\label{p:goal-anywhere} Reduction happens anywhere: a rule may fire in any position, under a
  $\lambda$, in a struct field and in an argument.
\item\label{p:goal-banf} A term in ANF (Section~\ref{sec:banf}) is a normal form: it
  has no step left, so reaching ANF never needs a rule that lifts a term into a $\kw{let}$.
\item\label{p:goal-agree} Every path agrees: if one reduction path ends in an ANF term, every path that ends,
  ends in that same term. This is confluence (\pref{p:confluent}).
\item\label{p:goal-ends} Confluence can't promise that every path ends:  some orders may loop while
  others finish, as in $\{a = \Omega, b = \hex{01}\}.b$, where $\Omega = (\lam{x}{x\ x})\ (\lam{x}{x\ x})$ steps to itself. It promises only that every path that ends gives the same normal form.
\item\label{p:goal-strategy} One order of steps, the strategy (Section~\ref{sec:strategy}), is complete and guaranteed: it
  chooses a step whenever the term has one, and if the term has a normal form, it reaches it.
\end{points}

%% ── Syntax ─────────────────────────────────────────────────────────────

\section{Kernel Syntax}\label{sec:syntax}

\begin{points}
\item The terms $t$ and two classes of them: values $v$ and data $D$.
\[
\begin{array}{l@{\quad}rcll}
 & t & ::= & & \textit{term} \\
\pt &   & \mid & x & \textit{variable} \\
\pt &   & \mid & \lam{x}{t} & \textit{abstraction} \\
\pt &   & \mid & t\ t & \textit{application} \\
\pt &   & \mid & b & \textit{byte array} \\
\pt &   & \mid & \{l_i = t_i\}_{i<k} & \textit{struct} \\
\pt &   & \mid & t.l & \textit{projection} \\
\pt &   & \mid & \kw{fix}\ t & \textit{fixed point} \\[6pt]
\pt & v & ::= & x \mid \lam{x}{t} \mid b \mid \{l_i = v_i\}_{i<k} \mid \kw{fix}\ (\lam{x}{t})  &
  \textit{value} \\
\pt\plabel{p:data} & D & ::= & b \mid \{l_i = D_i\}_{i<k} &
  \textit{data}
\end{array}
\]
\item A lambda $\lam{x}{t}$ extends as far right as possible:
  $\lam{x}{\lam{y}{t}}$ is $\lam{x}{(\lam{y}{t})}$. The dot right after the binder ends the binder; every other
  dot is projection: $\lam{p}{p.a}$ is $\lam{p}{(p.a)}$.
\item $\kw{fix}$ is a keyword. $\kw{fix}\ t$ parses as an application does:
  $\kw{fix}\ g.l\ a$ is $(\kw{fix}\ (g.l))\ a$.
\item A byte array $b$ is a sequence of bytes.
  \begin{subpoints}
  \item Where a number is expected, $b$ is read as an unsigned, little-endian
    number: $\hex{0201}$ is $258$, and the empty byte array is $0$.
  \item It is mostly written in hex, $\hex{01ff}$.
  \item When its bytes are valid UTF-8 text, it may be written as text
    instead: $\str{i32}$ is the byte array $\hex{693332}$.
  \end{subpoints}
\item A struct's fields are ordered.
  \begin{subpoints}
  \item $\{l_i = t_i\}_{i<k}$ has $k$ fields, $l_0 = t_0$ up to $l_{k-1} = t_{k-1}$; with
    $k = 0$ it is the empty struct $\{\}$.
  \item A label may be empty (no label); the other labels are unique.
  \end{subpoints}
\end{points}

%% ── Computation ────────────────────────────────────────────────────────

\section{Computation (at the root)}\label{sec:computation}

\begin{points}
\item The computation rules reduce a term at its root.
\begin{rules}
\axiomrow{E-Beta}
  {(\lam{x}{t_1})\ v \;\step\; \subst{x}{v}{t_1}}
\rulerow{E-Opcode}
  {\fn{op}(b)(D_i)_{i<k} = b'}
  {b\ \{l_i = D_i\}_{i<k} \;\step\; b'}
\rulerow{E-Proj}
  {l_j = l}
  {\{l_i = t_i\}_{i<k}.l \;\step\; t_j}
\rulerow{E-Index}
  {b < k}
  {\{l_i = t_i\}_{i<k}\ b \;\step\; t_b}
\rulerow{E-Default}
  {0 < k \le b}
  {\{l_i = t_i\}_{i<k}\ b \;\step\; t_{k-1}}
\axiomrow{E-FixProj}
  {(\kw{fix}\ (\lam{x}{t})).l \;\step\; (\subst{x}{\kw{fix}\ (\lam{x}{t})}{t}).l}
\axiomrow{E-FixApp}
  {(\kw{fix}\ (\lam{x}{t_1}))\ t_2 \;\step\; (\subst{x}{\kw{fix}\ (\lam{x}{t_1})}{t_1})\ t_2}
\end{rules}
\item Terms are equal up to $\alpha$-equivalence (renaming bound variables).
  When a bound variable of $t_1$ clashes with a free variable of $t_2$,
  $\subst{x}{t_2}{t_1}$ first renames it ($\alpha$-conversion), then
  substitutes.
\item\label{p:values} A value is a variable, $\lam{x}{t}$, a byte array, a struct of values or a fixed point $\kw{fix}\ (\lam{x}{t})$ (Section~\ref{sec:syntax}).
  \rn{E-Beta} substitutes only a value, so it never copies or drops a term that still has to
  run: in $(\lam{x}{\{x, x\}})\ (d.\mathit{print}\ \{r\})$ the call is not a value, so it stays as
  written, a $\kw{let}$ (\pref{p:let}), and runs once. A variable is a value because it stands for one.
  \begin{subpoints}
  \item\label{p:call} A call to a function, such as $d.\mathit{print}\ \{r\}$ (Section~\ref{sec:modules}),
    is an effect too. Reduction never binds a module's dependencies $d$: the linker connects them, so the call never
    reduces. It is not a value, so \rn{E-Beta} never copies or drops it: a $\kw{let}$ (\pref{p:let}) keeps it where it is written.
  \item A fixed point is a value: it has no effect, so copying it runs nothing twice, and it
    unfolds only when it is used (\pref{p:fix}).
  \end{subpoints}
\item\label{p:opcode} A byte array applied to a struct, $b\ \{l_i = t_i\}_{i<k}$, is an
  opcode call: $b$ names the opcode and the struct holds its operands, as in
  $\str{add}\ \{\str{i32}, a, b\}$. \rn{E-Opcode} computes the call when $\fn{op}(b)$ is
  defined: \[\str{add}\ \{\str{i32}, \hex{02000000}, \hex{03000000}\} \step \hex{05000000}\]
  \begin{subpoints}
  \item $\fn{op}(b)$ is a partial function from operand values to a byte array.
  \item Effects include memory opcodes, such as $\str{alloca}$, $\str{load}$ and $\str{store}$.
  \item Reduction has no memory: it never computes an address or the contents of
    one, and it never runs a memory opcode. It never moves a term that is not a value either
    (\pref{p:values}), so the order the program gives its effects survives into the normal form: in
    \[\begin{array}{l}\kw{let}\ u = \str{store}\ \{\str{i32}, \hex{05000000}, p\}\ \kw{in} \\
    \kw{let}\ x = \str{load}\ \{\str{i32}, p\}\ \kw{in} \\
    t\end{array}\]
    the $\str{store}$ stays outside the $\str{load}$, and the translation to LLVM emits the $\str{store}$
    first, so the $\str{load}$ reads $\hex{05000000}$. Using one's result in the next orders them too.
  \item All operands are data: byte arrays, or structs of data (row \pref{p:data}), never
    a lambda, so they never step and the result is the same in every reduction order.
  \item Otherwise the call does not reduce: an effect, undefined behavior, an unknown opcode, or an
    operand that is not data, such as a parameter $p.a$. With normal operands it is normal. The translation to LLVM emits it as an instruction.
    \end{subpoints}
\item \rn{E-Proj} takes the field labeled $l$ (labels are unique): $\{x = a, b, y = c\}.x \step a$.
  \begin{subpoints}
  \item The other fields are discarded, even one with an effect:
    $\{a = e, b = \hex{01}\}.b \step \hex{01}$. Reduction may
    reduce $e$ or not, but $e$ is not in the normal form: it is dropped.
  \item To run it, bind it first with a $\kw{let}$:
    $(\lam{x}{\{a = x, b = \hex{01}\}.b})\ e$.
    \end{subpoints}
\item \rn{E-Index}: a struct applied to a byte array $b$, with $b < k$, selects the field at position $b$.
  In $q = \{x = a, b, y = c\}$: $q\ \hex{00} = q.x = a$, $q\ \hex{01} = b$,
  $q\ \hex{02} = q.y = c$. An unlabeled field is reached only by its position.
  \begin{subpoints}
    \item\label{p:branch} A branch is a struct applied to a condition that is not a byte array yet; when it
    becomes one, \rn{E-Index} or \rn{E-Default} picks the arm.
    $\{\mathit{false} = t_2, \mathit{true} = t_1\}\ c$ takes $t_1$ when $c$ is
    not $0$ and $t_2$ when it is $0$.
  \item With more fields it is a multi-way branch.
  \item The labels only document the arms.
  \item It keeps one arm and discards the others. Reduction
    may reduce a discarded arm or not, but it does not contribute to the normal form.
  \end{subpoints}
\item \rn{E-Default}: past the last field, a byte array selects the last one, so the last field is
  a default arm: $q\ \hex{05} = c$. The empty struct has no default (\pref{p:stuck}).
\item\label{p:fix} A fixed point $\kw{fix}\ (\lam{x}{t})$ is recursion on any value, a struct
  too: it stands for $t$, with $x$ standing for the fixed point itself.
  \begin{subpoints}
  \item It unfolds only when it is used: projected (\rn{E-FixProj}) or applied (\rn{E-FixApp}).
    Each use unfolds it once, putting $\kw{fix}\ (\lam{x}{t})$ in for $x$; on its own it never
    steps at the root.
  \item On a struct: in $\kw{fix}\ (\lam{f}{\{l = \lam{p}{f.l\ p}\}})$, $f$ is the fixed point
    itself, so $f.l$ unfolds it and takes the field $l$.
  \item An unfolded copy holds the fixed point again, often under a lambda, where reduction may
    unfold it again and again. Every path that ends gives the same
    normal form (\pref{p:confluent}); a normal form holds a fixed point only where it is not used.
  \item For example, with numbers in decimal standing for $\str{i32}$ byte arrays,
    \[\begin{array}{l}F = \kw{fix}\ (\lam{f}{\lam{n}{\kw{if}\ \str{icmp}\ \{\str{eq}, \str{i32}, n, 0\}\ \kw{then}\ 1}} \\
\qquad \kw{else}\ \str{mul}\ \{\str{i32}, n, f\ (\str{sub}\ \{\str{i32}, n, 1\})\})\end{array}\]
    every path of $F\ 3$ that ends gives $6$. $F\ p.n$ has no normal form: recursion on an
    unknown never stops. Recursion in LLVM goes through $s$ and groups (Section~\ref{sec:banf}).
  \item For example, $\kw{fix}$ binds $s$ and groups to run an ingot by reduction (\pref{p:run-module}).
  \end{subpoints}
\end{points}

%% ── Congruence ─────────────────────────────────────────────────────────

\section{Reduction Anywhere}\label{sec:congruence}

\begin{points}
\item The congruence rules reduce a term inside it.
\begin{rules}
\rulerow{E-Abs}{t \step t'}{\lam{x}{t} \;\step\; \lam{x}{t'}}
\rulerow{E-App1}{t_1 \step t_1'}{t_1\ t_2 \;\step\; t_1'\ t_2}
\rulerow{E-App2}{t_2 \step t_2'}{t_1\ t_2 \;\step\; t_1\ t_2'}
\rulerow{E-Struct}{t_i \step t_i'}
  {\{\dots, l_i = t_i, \dots\} \;\step\; \{\dots, l_i = t_i', \dots\}}
\rulerow{E-Proj1}{t \step t'}{t.l \;\step\; t'.l}
\rulerow{E-Fix1}{t \step t'}{\kw{fix}\ t \;\step\; \kw{fix}\ t'}
\end{rules}
\item $\step$ is reduction anywhere: a rule of Section~\ref{sec:computation} may fire
  anywhere, under a lambda, in a struct field (so in every arm of a branch) too.
  A struct's fields are computed in no particular order.
  $t \steps t'$ is zero or more $\step$ steps.
\item\label{p:confluent} $\step$ is confluent: it is the call-by-value $\lambda$-calculus (Plotkin's $\lambda_v$) with reduction anywhere, with
   structs, projection, branching and a fixed point (Church--Rosser; projection is the usual record rule). The
  proof is in Lean 4, in \texttt{proofs/} (theorem \texttt{Oymomo.confluence}), for
  every opcode table $\fn{op}$.
  \begin{subpoints}
  \item Paths that end agree, even when other paths loop: if $t \steps n_1$ and $t \steps n_2$ with
    both normal, confluence gives a $u$ with $n_1 \steps u$ and $n_2 \steps u$; a normal form has
    no step, so $n_1 = u = n_2$. Looping paths don't matter: confluence never asks every
    path to end.
    $\kw{fix}$ adds \rn{E-FixProj}, \rn{E-FixApp} and \rn{E-Fix1} (\pref{p:fix}); the proof covers them.
  \item Confluence is what lets programs be combined. For two programs $A$ and $B$ merged as
    $A\ B$: reducing $A$ and $B$ first, $A \steps A'$ and $B \steps B'$, and then $A\ B$ is one of the orders of $A\ B$
    (\rn{E-App1}, \rn{E-App2}: $A\ B \steps A'\ B'$), so it reaches the same normal form as applying first. Merging never binds a
    module's $d$ (\pref{p:call}), so a call between $A$ and $B$ is never inlined.
    \end{subpoints}
\end{points}
\begin{points}
\item\label{p:normal} A normal form is a term with no step. By confluence (\pref{p:confluent}), a term has at most one.
\item\label{p:stuck} A normal form is stuck when it contains a part that no substitution can make
  reduce, such as $\{\}\ t$, a byte array applied to a lambda, or $t.l$ where $t$ is a lambda, or $\kw{fix}\ t$ where $t$ is a byte array or a struct. A
  program $t$ with $t \steps t'$ for a stuck $t'$ is rejected.
\item\label{p:forms} A normal form that is not stuck is valid. An unknown opcode or undefined behavior is not stuck: the call is normal (\pref{p:opcode}). A normal form is in ANF (Section~\ref{sec:banf}) or
  not; only one in ANF is translated to LLVM. $\str{add}\ \{\str{i32}, \str{mul}\ \{\dots\}, x\}$
  is normal but not in ANF; written with a  $\kw{let}$ for the $\str{mul}$, it is.
\end{points}


%% ── ANF and Strategy ─────────────────────────────────────────────────

\section{Strategy}\label{sec:strategy}

\begin{points}
\item\label{p:strategy} If a term has a normal form, any order of steps that ends reaches it
  (\pref{p:confluent}).
\item\label{p:order} This order is one way. Each row matches a shape and chooses either a rule of
  Section~\ref{sec:computation}, which fires at the root, or a part of the term, which takes the next step
  by the rule of Section~\ref{sec:congruence} shown; the order defines no rule of its own. Rows are tried top to bottom, and a part is chosen only when it has a step.
  A term matches rows of only one group (projection, application, struct, lambda, fixed point), so only the
  order within a group matters.
\[
\begin{array}{l@{\quad}l@{\quad}l}
 & \textit{shape} & \textit{choose} \\ \hline
\pt & \{l_i = t_i\}_{i<k}.l \quad l_j = l & \rn{E-Proj} \\
\pt & (\kw{fix}\ (\lam{x}{t})).l & \rn{E-FixProj} \\
\pt\plabel{p:order-proj} & t.l & \rn{E-Proj1}\text{ on }t \\
\pt & (\lam{x}{t_1})\ v & \rn{E-Beta} \\
\pt\plabel{p:order-arg} & (\lam{x}{t_1})\ t_2 & \rn{E-App2}\text{ on }t_2 \\
\pt & (\kw{fix}\ (\lam{x}{t_1}))\ t_2 & \rn{E-FixApp} \\
\pt & b\ \{l_i = D_i\}_{i<k} \quad \fn{op}(b)(D_i)_{i<k} = b' & \rn{E-Opcode} \\
\pt & \{l_i = t_i\}_{i<k}\ b \quad b < k & \rn{E-Index} \\
\pt & \{l_i = t_i\}_{i<k}\ b \quad 0 < k \le b & \rn{E-Default} \\
\pt\plabel{p:order-cond} & \{l_i = t_i\}_{i<k}\ t & \rn{E-App2}\text{ on }t \\
\pt & t_1\ t_2 & \rn{E-App1}\text{ on }t_1 \\
\pt & t_1\ t_2 & \rn{E-App2}\text{ on }t_2 \\
\pt\plabel{p:order-field} & \{l_i = t_i\}_{i<k} & \rn{E-Struct}\text{ on the first } t_i \text{ not a value} \text{ with a step} \\
\pt & \{l_i = t_i\}_{i<k} & \rn{E-Struct}\text{ on the first } t_i \text{ with a step} \\
\pt\plabel{p:order-lam} & \lam{x}{t} & \rn{E-Abs}\text{ on }t \\
\pt & \kw{fix}\ t & \rn{E-Fix1}\text{ on }t
\end{array}
\]
\item What the order gives:
  \begin{subpoints}
  \item A projected term is reduced before its fields (row \pref{p:order-proj}); \rn{E-Proj} keeps one
    field and the others are never reduced.
  \item An argument is reduced to a value before the function's body (row \pref{p:order-arg}).
  \item A branch's condition is reduced before its arms (row \pref{p:order-cond}); when it is a byte array,
    \rn{E-Index} or \rn{E-Default} keeps one arm and the others are never reduced.
  \item The order goes under a lambda only when nothing outside it can step (row \pref{p:order-lam}), so a fixed point inside a lambda unfolds only
    when nothing outside can step (\pref{p:fix}). In a
    struct, a field that is not a value is reduced before a field that is (row \pref{p:order-field}):
    $(\lam{x}{x.g})\ \{f = \lam{u}{\Omega}, g = (\lam{y}{y})\ \hex{01}\}$ reduces $g$ and never enters $\lam{u}{\Omega}$.
  \end{subpoints}
\item Each step of this order is a $\step$ step, and when no row applies the term has no step
  (completeness), so when the order ends, it ends in the normal form (\pref{p:normal}). If a term has a
  normal form, the order reaches it (guarantee). Both are part of the goal (\pref{p:goal-strategy}). In Lean 4, in \texttt{proofs/}, the order is a relation with one rule per row of the table;
  each rule says that the row's shape matches, that the chosen part has a step, and that no
  earlier row of its group applies. Completeness and the guarantee matter for an implementation that follows
  this order: it never stops early, and it never loops when a normal form exists.
  \begin{subpoints}
  \item Soundness (theorem \texttt{Oymomo.sstep\_step}): each step of the order is a $\step$ step, so the
    order defines no rule of its own. Each row's rule is a rule of Section~\ref{sec:computation} or~\ref{sec:congruence}.
  \item Completeness (theorem \texttt{Oymomo.sstep\_complete}): if a term has a $\step$ step, the order chooses
    one. The proof goes by the shape of the term: a rule at the root is a row of its own, and
    otherwise the step is inside a part, which the order reaches by the rows that choose a part.
  \item Determinism (theorem \texttt{Oymomo.sstep\_det}): the order chooses at most one step, since the rows of
    a group exclude each other. With completeness, every term with a step gets exactly one choice.
  \item Guarantee (theorem \texttt{Oymomo.guarantee}): if some steps lead from $t$ to a normal form $n$,
    the order leads from $t$ to $n$ too, so it never loops when a normal form exists. The proof shows that a
    step the order does not choose can always wait: taking the order's step first still reaches $n$.
  \end{subpoints}
\end{points}

%% ── ANF and Functions ────────────────────────────────────────────────

\section{ANF and Functions}\label{sec:banf}

\begin{points}
\item\label{p:definition} A function's body is $\lam{d}{\lam{s}{\lam{p}{G}}}$: $d$ holds the module's dependencies,
  $s$ its own symbols and $p$ the arguments, and $G$ is a group: nested groups of blocks, each block's body
  an expression in ANF.
\item\label{p:anf} An expression in ANF is a list of $\kw{let}$s, one per instruction, ending in a terminal;
  a terminal may be a group, which nests more blocks. Every variable is written $x$: the grammar does not
  tell variables apart, and the verifier does (\pref{p:kinds}).
\[
\begin{array}{l@{\quad}rcll}
\pt\plabel{p:path} & P & ::= & x \mid P.l & \textit{path} \\
\pt & O & ::= & P \mid b & \textit{operator} \\
\pt & Q & ::= & P \mid D & \textit{operand} \\
\pt & J & ::= & P\ Q_1 \cdots Q_n & \textit{jump} \\
\pt & T & ::= & Q \mid J \mid \{J_i\}_{i<k}\ P \mid G & \textit{terminal} \\
\pt & E & ::= & \kw{let}\ x = O\ \{l_i = Q_i\}_{i<k}\ \kw{in}\ E \mid T & \textit{expression} \\
\pt & K & ::= & E \mid \lam{x_1}{\cdots \lam{x_n}{E}},\ 0 < n & \textit{block} \\
\pt & G & ::= & x\ \hex{}\ (\lam{x}{\{l_i = K_i\}_{i<k}}),\ 0 < k & \textit{group}
\end{array}
\]
\item What each class is:
  \begin{subpoints}
  \item $P$: a path is a variable or a path and a label: $p.a$, $s.\mathit{gcd}$, $d.\mathit{puts}$, $g.\mathit{loop}$. The
    verifier says where a path may have more than one label (\pref{p:kinds}). A field of a struct value is
    read with the opcode $\str{extractvalue}$ (\pref{p:struct-values}), never with a projection.
  \item $O$: an operator is a path to a function or an opcode $b$. Applied to the struct of its operands,
    $O\ \{l_i = Q_i\}_{i<k}$, it is an instruction: a call (\pref{p:calls}), or an opcode call that \rn{E-Opcode} does
    not compute (\pref{p:opcode}). The struct holds the operands; it is not a struct value.
  \item $Q$: an operand is a path or data (row \pref{p:data}). A path to a symbol stands for its address:
    $\str{load}\ \{\str{i32}, s.\mathit{counter}\}$ reads a global variable. A struct operand is a constant, such as
    the type $\{\str{i8}, \str{i32}\}$ in $\str{load}\ \{\{\str{i8}, \str{i32}\}, x\}$.
  \item $J$: a jump $P\ Q_1 \cdots Q_n$ goes to the block $P$, passing one operand for each of its $n$
    parameters; a block with no parameters is jumped to by $P$ alone.
  \item $T$: a terminal is a return $Q$, a jump $J$, a branch $\{J_i\}_{i<k}\ P$ whose arms all jump, or a
    group $G$. A call that ends a function is bound and returned: $\kw{let}\ y = s.f\ \{\dots\}\ \kw{in}\ y$.
  \item $E$: an expression is the $\kw{let}$s of one block, up to its terminal.
  \item $K$: a block is an expression, or a $\lambda$ taking the block's parameters one by one.
  \item $G$: a group $s\ \hex{}\ (\lam{g}{\{l_i = K_i\}_{i<k}})$ names blocks: $g.l_i$ is the block $K_i$. $s\ \hex{}$,
    field $0$ of the module's instance, is the runner of groups (\pref{p:run-module}). Field $0$ of the group
    is an expression, never a $\lambda$.
  \end{subpoints}
\item\label{p:kinds} Each variable has a kind, given by its binder; the verifier checks every use of a
  variable against its kind and rejects the others:
\[
\begin{array}{l@{\quad}lll}
 & \textit{kind} & \textit{bound by} & \textit{used as} \\ \hline
\pt & \textit{dependencies} & \text{the body's 1st } \lambda,\ d & d.l,\ l \text{ in } \mathit{depends}\text{: operator, operand} \\
\pt & \textit{self} & \text{the body's 2nd } \lambda,\ s & s.l,\ l \text{ in } \mathit{symbols}\text{: operator, operand};\ s\ \hex{}\text{: runner} \\
\pt & \textit{arguments} & \text{the body's 3rd } \lambda,\ p & p.l,\ l \text{ a parameter: operand, condition} \\
\pt & \textit{group} & \text{a group's } \lambda,\ g & g.l,\ l \text{ a labeled field: head of a jump} \\
\pt & \textit{value} & \kw{let},\ \text{a block's } \lambda & x\text{: operand, condition}
\end{array}
\]
  \begin{subpoints}
  \item $d.l$ and $s.l$ are operators when $l$ is a function, and otherwise operands, the address of $l$.
    $s\ \hex{}$ is only the head of a group; any other $x\ b$ is rejected.
  \item A jump $g.l\ Q_1 \cdots Q_n$ passes exactly as many operands as the block $l$ has parameters.
  \item A value has no fields: $x.l$ with $x$ a value is rejected.
  \item What the grammar leaves open, the kind decides: a path alone as a terminal is a jump when its head is
    a group, and a return otherwise.
  \end{subpoints}
\item\label{p:groups} Where each binder's variables are seen:
  \begin{subpoints}
  \item The body's $d$, $s$ and $p$ are seen in all of its group.
  \item $\kw{let}\ x = O\ \{l_i = Q_i\}_{i<k}\ \kw{in}\ E$ adds the value $x$ to $E$.
  \item A group adds its $g$ to all its fields, so its blocks may jump to each other and to themselves.
  \item A block $\lam{x_1}{\cdots \lam{x_n}{E}}$ adds the values $x_1, \dots, x_n$ to $E$.
  \item So a block sees the let variables bound around it: a block of a group runs only after the code
    before the group, and nested groups give dominance.
  \end{subpoints}
\item\label{p:top-group} Where each block starts:
  \begin{subpoints}
  \item The body's group is the top group: its field $0$ is the entry block, and is labeled.
  \item Any other group continues the block it ends: its field $0$ is unlabeled and runs on after the
    $\kw{let}$s before the group; its other fields are new blocks.
  \item No jump goes to a field $0$: no path names it.
  \end{subpoints}
\item\label{p:calls} A call is $P\ \{x_i = Q_i\}_{i<k}$, with $P$ a path $s.l$ or $d.l$ to a function and $x_i$ its
  parameters' labels, in order. A path to a function that is not called is the function's address.
\item\label{p:struct-values} A struct value, such as a $\str{load}$ of a struct type, is read and built
  with opcodes, as in LLVM, with positions for indices:
  \[\begin{array}{l}\kw{let}\ s_1 = \str{insertvalue}\ \{\{\str{i8}, \str{i32}\}, \{\hex{00}, \hex{00000000}\}, p.a, \hex{00}\}\ \kw{in} \\
    \kw{let}\ s_2 = \str{insertvalue}\ \{\{\str{i8}, \str{i32}\}, s_1, w, \hex{01}\}\ \kw{in} \\
    \kw{let}\ y = \str{extractvalue}\ \{\{\str{i8}, \str{i32}\}, s_2, \hex{01}\}\ \kw{in}\ \dots\end{array}\]
\item\label{p:anf-rules} What ANF rules out, so that each part is one instruction:
  \begin{subpoints}
  \item No nested instruction: $\str{add}\ \{\str{i32}, \str{mul}\ \{\dots\}, x\}$ is normal (\pref{p:forms}) but not in
    ANF. The fix is the program's: bind the $\str{mul}$ with a $\kw{let}$. No rule lifts it.
  \item No nested $\kw{let}$ and no branch as a value: the arms of a branch only jump.
  \item No struct value built from parts that are not constants: it is built with the opcode
    $\str{insertvalue}$ (\pref{p:struct-values}).
  \end{subpoints}
\item\label{p:anf-normal} A body is a normal form (\pref{p:normal}), as Goal \pref{p:goal-banf} asks:
  \begin{subpoints}
  \item An instruction $O\ \{l_i = Q_i\}_{i<k}$ is not a value, so \rn{E-Beta} never fires on its $\kw{let}$.
  \item A path or a jump has a variable at its head, and an operand is never applied or projected, so no
    other rule fires at them.
  \item A branch applies a struct to a path, never to a byte array, so \rn{E-Index} and \rn{E-Default} don't fire.
  \item A group applies $s\ \hex{}$, and nothing applies the body, so $s$ stays free and the group doesn't reduce.
  \item The proof is in Lean 4, in \texttt{proofs/} (theorem \texttt{Oymomo.anf\_normal}), for expressions
    without groups; groups are not in it yet.
  \end{subpoints}
\item\label{p:gcd-body} For example, the body of Euclid's $\gcd$, with the sugar of \pref{p:if}. The block
  $\mathit{loop}$ holds a nested group: its field $0$ goes on after the $\kw{let}$ and branches, and $\mathit{step}$ is a
  new block that uses $a$ and $b$ without being passed them:
  \begin{Verbatim}[commandchars=\@\<\>, xleftmargin=2em]
@vlamd. @vlams. @vlamp. s [] (@vlamg. {
  entry = g.loop p.a p.b,
  loop = @vlama. @vlamb.
    let z = 'icmp' {'eq', 'i32', b, [00000000]} in
    s [] (@vlamh. {
      if z then g.done a else h.step,
      step = let r = 'urem' {'i32', a, b} in g.loop b r
    }),
  done = @vlamr. r
})
\end{Verbatim}
\end{points}

%% ── Ingots ───────────────────────────────────────────────────────────

\section{Ingots}\label{sec:modules}

\begin{points}
\item\label{p:module} An ingot is a module in the shape LLVM wants, so that its translation
  (Section~\ref{sec:llvm}) only reads it: $\{\mathit{version} = b,\ \mathit{depends} = \{l_i = t_i\}_{i<n},\ \mathit{symbols} = \{l_i = e_i\}_{i<k},\ \mathit{target} = t\}$.
  \begin{subpoints}
  \item A program reduces to an ingot; only an ingot is translated, and one ingot is one LLVM module.
  \item $\mathit{version}$ names the shape of this section, as text: $\str{0}$. Versions are compared only for
    equality; an unknown version is rejected.
  \item $\mathit{depends}$ lists what the module uses from other modules: each label is a symbol's name and
    each value its type (\pref{p:types}, \pref{p:entry-kind}), such as
    $\mathit{puts} = \{\str{function}, \{s = \str{ptr}\}, \str{i32}\}$.
  \item $\mathit{symbols}$ holds the module's own symbols, each an entry $e_i$ labeled by the symbol's name.
  \item $\mathit{target}$ may be left out; it is read later, with the target's details. Any other field is
    rejected.
  \item Each entry is closed: its body names other symbols only through $d$ and $s$ (\pref{p:kinds}), so a
    source language lowers each of its items to one entry on its own.
  \end{subpoints}
\item\label{p:entry-kind} An entry is a struct with a $\mathit{type}$ and exactly one of
  $\mathit{body}$, $\mathit{init}$, $\mathit{value}$; that field gives its kind:
\[
\begin{array}{l@{\quad}l@{\quad}l@{\quad}l}
 & \textit{fields} & \textit{kind} & \textit{type} \\ \hline
\pt & \mathit{type}, \mathit{body} & \textit{definition} & \{\str{function}, \{x_i = t_i\}_{i<k}, t\} \\
\pt & \mathit{type}, \mathit{init} & \textit{global variable} & \text{a type (\pref{p:types})} \\
\pt & \mathit{type}, \mathit{value} & \textit{constant} & \text{a type}
\end{array}
\]
  \begin{subpoints}
  \item A definition's $\mathit{body}$ is a function's body (Section~\ref{sec:banf}).
  \item A function's type holds its parameters' types, in order, and its result type:
    $\mathit{puts}$ is $\{\str{function}, \{s = \str{ptr}\}, \str{i32}\}$.
  \item A name in both $\mathit{depends}$ and $\mathit{symbols}$ is rejected.
  \end{subpoints}
\item\label{p:init} An $\mathit{init}$ or a $\mathit{value}$ is data (row \pref{p:data}) of the entry's type:
  $\{\mathit{type} = \str{i32},\ \mathit{init} = \hex{29000000}\}$.
  \begin{subpoints}
  \item Where the type is $\str{ptr}$, a byte array names a symbol and stands for its address, and
    $\str{}$ is the null pointer; $\{b_1, b_2\}$ is the symbol $b_1$ plus $b_2$ bytes.
  \item An array of $\str{i8}$ may be one byte array, its bytes in order:
    $\{\mathit{type} = \{\str{array}, \hex{03}, \str{i8}\},\ \mathit{value} = \hex{686900}\}$.
  \item $\{\}$ is all zeros, for any type.
  \end{subpoints}
\item\label{p:run-module} A module runs by reduction too, as a test does, when $d$ and $s$ are bound. For a
  module with definitions $f_i$ of bodies $B_i$, and a struct $u$ with a field for each name in its
  $\mathit{depends}$, the instance is
  \[\kw{fix}\ (\lam{s}{\{\lam{G}{(\kw{fix}\ G)\ \hex{00}},\ f_i = B_i\ u\ s\}_{i<k}})\]
  \begin{subpoints}
  \item Projecting $f_i$ unfolds the instance (\rn{E-FixProj}) and takes $B_i\ u\ s$ with $s$ the instance, so a
    call of $f_i$ runs its body; a call $s.f_j\ \{\dots\}$ inside it reaches the instance again.
  \item Field $0$, unlabeled, is the runner of groups: the module runs its own groups, and no dependency
    supplies the runner. $s\ \hex{}$ unfolds the instance (\rn{E-FixApp}) and takes field $0$ (\rn{E-Index}), so a
    group $s\ \hex{}\ (\lam{g}{t})$ becomes $(\kw{fix}\ (\lam{g}{t}))\ \hex{00}$, which runs field $0$ of $t$ with $g$
    bound to its own blocks; a jump $g.l\ Q_1 \cdots Q_n$ takes the block $l$ and applies it.
  \item Dependencies are matched by name, so instantiating modules in dependency order gives each module
    one instance of each module below it; in a diamond, both sides share the same instance.
  \item With the module of \pref{p:gcd} and $u = \{\}$, a call of the instance's $\mathit{gcd}$ on
    $\{a = \hex{30000000}, b = \hex{12000000}\}$ reduces to $\hex{06000000}$.
  \item An opcode that \rn{E-Opcode} does not compute, such as $\str{load}$, stays in the
    result, waiting for run time.
  \end{subpoints}
\item\label{p:merge} Modules are combined by user code; this document gives no rule for it:
  \begin{subpoints}
  \item To run by reduction, a program instantiates its modules (\pref{p:run-module}) and passes each the
    instances it depends on. A renaming is a struct of paths, such as $\{g = s.h\}$; it reduces once $s$ is
    bound.
  \item To compile, each module is its own ingot and its own LLVM module, and the system linker joins them
    by name: a name in $\mathit{depends}$ to the symbol of that name in another module.
  \end{subpoints}
\item\label{p:gcd} For example, an ingot with $\gcd$ (its body is \pref{p:gcd-body}) and $\mathit{main}$:
  \begin{Verbatim}[commandchars=\@\<\>, xleftmargin=2em]
{
  version = '0',
  depends = {},
  symbols = {
    gcd = {type = {'function', {a = 'i32', b = 'i32'}, 'i32'},
           body = @vlamd. @vlams. @vlamp. s [] (@vlamg. { @ldots })},
    main = {type = {'function', {}, 'i32'},
            body = @vlamd. @vlams. @vlamp. s [] (@vlamg. {
              entry = let r = s.gcd {a = [30000000], b = [12000000]} in r
            })}
  }
}
\end{Verbatim}
\end{points}

\section{LLVM Translation}\label{sec:llvm}

\begin{points}
\item\label{p:types}
  Oymomo has no types and no type checking: it is an untyped $\lambda$-calculus. An LLVM type is an
  ordinary term that only an opcode or an entry reads, as an operand: $\str{add}\ \{\str{i32}, a, b\}$ is LLVM's
  $\kw{add\ i32}\ a, b$.
  A type is data (row \pref{p:data}) of one of these shapes, with $t_i$ types and $n$ a byte array read as a number:
  \begin{subpoints}
  \item $\str{i}N$, such as $\str{i1}$, $\str{i8}$ or $\str{i32}$: LLVM's integer type $\kw{i}N$.
  \item $\str{ptr}$: LLVM's opaque pointer.
  \item $\{l_i = t_i\}_{i<k}$: LLVM's struct $\{t_0, \dots, t_{k-1}\}$. The labels only name the fields.
  \item $\{\str{array}, n, t\}$ and $\{\str{vector}, n, t\}$: LLVM's $[n \times t]$ and $\langle n \times t\rangle$.
  \item $\{\str{packed}, t_0, \dots, t_{k-1}\}$: LLVM's packed struct $\langle\{t_0, \dots, t_{k-1}\}\rangle$.
  \item A struct whose field $0$ is $\str{array}$, $\str{vector}$, $\str{packed}$ or $\str{function}$ is read by these
    shapes or as a function's type (\pref{p:entry-kind}), never as a plain struct.
  \end{subpoints}
\item\label{p:bytes-type} A byte array has no type of its own either: it takes the type its use needs: the
  opcode's type operand, the parameter it is passed to, the result type of the function it returns from, or
  the $\mathit{type}$ of the entry it initializes. As an $\str{i}N$ it has $\lceil N/8 \rceil$ bytes:
  $\str{add}\ \{\str{i32}, \hex{01000000}, p.a\}$, never $\hex{01}$.
  \begin{subpoints}
  \item A count or index of $\str{alloca}$ or $\str{getelementptr}$ has no type operand: a byte array of $8$ bytes there is an
    $\str{i64}$, any other an $\str{i32}$.
  \end{subpoints}
\item\label{p:infer} A block's parameters have no written type: each takes the type of the operands
  jumped to it and of its uses, repeated until nothing changes.
  \begin{subpoints}
  \item A used parameter with no type found, or with two, is rejected: ``give $x$ a typed argument or a
    typed use''.
  \item An operand whose type differs from the type its use needs is rejected, at the operand.
  \end{subpoints}
\item\label{p:one-to-one} The translation is one to one: every LLVM block, label and instruction is
  written in the ingot, and the translation adds none and changes none.
  \begin{subpoints}
  \item Groups are flattened, in source order: each labeled field is a block. The runner $s\ \hex{}$ itself becomes nothing.
  \item Each $\kw{let}$ becomes one instruction: an opcode call becomes the opcode's instruction, and a
    call $s.l\ \{\dots\}$ or $d.l\ \{\dots\}$ becomes $\kw{call}$ of the function $l$.
  \item A return becomes $\kw{ret}$, and a jump $\kw{br}$.
  \item A branch of two arms on an $\str{i1}$ becomes $\kw{br\ i1}$, to arm $1$ when true and arm $0$ when false;
    any other branch becomes $\kw{switch}$, arm $i$ the case $i$ and the last arm the default.
  \item A block's parameters become its $\kw{phi}$ instructions, one incoming value per jump to it, even
    when only one jump goes there. Two arms of one branch that jump to the same block pass the same
    operands; otherwise the branch is rejected.
  \end{subpoints}
\item\label{p:names} Names: terms are equal up to renaming bound variables, so the translation reads
  each definition with all its bound variables named apart, a repeated name taking the smallest free
  number ($r$, $r1$).
  \begin{subpoints}
  \item An argument is named $p.x_i$, a block $g.l$ by its group and label, and a let variable by its name.
    No two can be equal.
  \item For \pref{p:gcd}, $\mathit{done}$'s parameter is named $r1$, and the blocks are $g.\mathit{entry}$,
    $g.\mathit{loop}$, $h.\mathit{step}$ and $g.\mathit{done}$.
  \end{subpoints}
\item\label{p:symbols-llvm} Each symbol and each name in $\mathit{depends}$ becomes an LLVM global named by its label, as
  $\kw{@gcd}$ or $\kw{@puts}$, with LLVM's default linkage:
  \begin{subpoints}
  \item A definition becomes LLVM's $\kw{define}$; its parameters and result come from its type.
  \item A function in $\mathit{depends}$ becomes $\kw{declare}$: $\mathit{puts} = \{\str{function}, \{s = \str{ptr}\}, \str{i32}\}$ is
    $\kw{declare\ i32\ @puts(ptr)}$; a variable in $\mathit{depends}$ becomes $\kw{external\ global}$.
  \item A constant becomes $\kw{@answer\ =\ constant\ i32\ 42}$, and a global variable
    $\kw{@counter\ =\ global\ i32\ 41}$; $\{\}$ becomes $\kw{zeroinitializer}$, and a byte array for an
    $\str{i8}$ array becomes $\kw{c"\dots"}$.
  \end{subpoints}
\end{points}
